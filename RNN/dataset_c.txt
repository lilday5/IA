int factorial(int n){
if(n <= 1){
return 1;
}
return n * factorial(n - 1);
}

int fibonacci(int n){
if(n <= 1){
return n;
}
return fibonacci(n - 1) + fibonacci(n - 2);
}

int sumaArreglo(int arreglo[], int n){
int suma = 0;
int i;
for(i = 0; i < n; i++){
suma += arreglo[i];
}
return suma;
}

int maximoArreglo(int arreglo[], int n){
int i;
int mayor = arreglo[0];
for(i = 1; i < n; i++){
if(arreglo[i] > mayor){
mayor = arreglo[i];
}
}
return mayor;
}

int minimoArreglo(int arreglo[], int n){
int i;
int menor = arreglo[0];
for(i = 1; i < n; i++){
if(arreglo[i] < menor){
menor = arreglo[i];
}
}
return menor;
}

int buscarElemento(int arreglo[], int n, int valor){
int i;
for(i = 0; i < n; i++){
if(arreglo[i] == valor){
return i;
}
}
return -1;
}

int contarMayores(int arreglo[], int n, int valor){
int i;
int contador = 0;
for(i = 0; i < n; i++){
if(arreglo[i] > valor){
contador++;
}
}
return contador;
}

int contarMenores(int arreglo[], int n, int valor){
int i;
int contador = 0;
for(i = 0; i < n; i++){
if(arreglo[i] < valor){
contador++;
}
}
return contador;
}

int i;
int contador = 0;
for(i = 0; texto[i] != '\0'; i++){
if(texto[i] == letra){
contador++;
}
}
return contador;
}

int i;
for(i = 0; texto[i] != '\0'; i++){
if(texto[i] == letra){
return 1;
}
}
return 0;
}
